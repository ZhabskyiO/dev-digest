// server/src/modules/badges/routes.ts  (new module in a PR)
import type { FastifyInstance } from "fastify";
import { eq, desc } from "drizzle-orm";
import { db } from "../../db/client.js";
import { reviews, findings } from "../../db/schema.js";

export default async function badgesRoutes(app: FastifyInstance) {
  app.get("/repos/:id/badge", async (req) => {
    const { id } = req.params as { id: string };
    const [latest] = await db
      .select()
      .from(reviews)
      .where(eq(reviews.prId, id))
      .orderBy(desc(reviews.createdAt))
      .limit(1);
    const rows = latest
      ? await db.select().from(findings).where(eq(findings.reviewId, latest.id))
      : [];
    return { score: latest?.score ?? null, findings: rows.length, color: rows.length > 3 ? "red" : "green" };
  });
}
